= Formatted attribute ordering

:copyright: (C)
:target: https://example.org
:name: Early
:quotes-first: pass:q,a[{strong}]
:attributes-first: pass:a,q[{strong}]
:wrapped: pass:q,a[*{name}*]
:replacements-first: pass:r,a[{copyright}]
:replacements-after: pass:a,r[{copyright}]
:macros-first: pass:m,a[{target}]
:macros-after: pass:a,m[{target}]
:normal: pass:normal[{strong}]
:alias: {quotes-first} / {attributes-first}

:name: Late
:strong: *Changed*
:target: https://changed.example
