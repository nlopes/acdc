// I05 parent after include

[listing]

[literal]

include::includes/styled_comment_boundary.adoc[]
// HIDDEN parent ordinary comment
I12 ordinary after include

include::includes/styled_comment_open.adoc[]
// I14 parent after included style

[source,text]

// I16 still in parent code

[source,text]

include::includes/styled_comment_nested.adoc[]
// I18 after nested include

I19 ordinary end
